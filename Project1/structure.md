#part3 (automate that process of running the application on server01 using (systemd services)
```
just go to server01
ssh -p 2221 admin@localhost
#now go to the system folder on systemd
cd /etc/systemd/system
vi blog.service
and put these bash script
[Unit]
Description=My Application
After=network.target

[Service]
User=admin
Group=admin
WorkingDirectory=/opt/myapp/Blog_bost_with_ArgoCd/Project1/Project1_metrics_logs_traces_Argocd/software
ExecStart=/opt/myapp/.blog/bin/python3 /opt/myapp/Blog_bost_with_ArgoCd/Project1/Project1_metrics_logs_traces_Argocd/software/run.py
Restart=on-failure
[Install]
WantedBy=multi-user.target

#now restart systemd
systemctl daemon-reload
systemctl enable blog.service
systemctl start blog.service
# now just try to access that app again using that url: http://ip-of-server01:8080
```
